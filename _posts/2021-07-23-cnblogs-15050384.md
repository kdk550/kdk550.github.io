```
//选择排序
#include<iostream>
using namespace std;
int main()
{
    int n;
    cin >> n;
    int a[1010];
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    for (int i = 1; i <= n-1; i++)
    {
        for (int j = i + 1; j <=n; j++)
        {
            if (a[i] > a[j])
            {
                int temp = a[j];
                a[j] = a[i];
                a[i] = temp;
            }
        }
    }
    for (int i = 1; i <= n; i++)
        cout << a[i]<<"   ";
    return 0;
}
```



 选择排序的算法时间复杂度为O（n^2）

空间复杂度为O（n）
