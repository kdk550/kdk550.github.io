H  
首先观察到deg权值的数量很少，只需要记录deg权值为<span class="math inline">\(x\)</span>的数量，再与其他权值<span class="math inline">\(x'\)</span>做那个公式的值乘<span class="math inline">\(x\)</span>的数量乘<span class="math inline">\(x'\)</span>的数量即可

我的作法离散化了一下



```cpp
typedef long long ll;
const ll mod = 998244353;
const int N = 1e6 + 10;
int n, m, cnt;
ll d[N];
vector<int> b;
map<int, int> mp;
ll s[4010];
ll val[4010][4010];
ll id[4000010];
void solve()
{
    cin>>n>>m;
    for(int i = 1; i <= m; i++)
    {
        int u, v;   cin>>u>>v;
        d[u]++, d[v]++;
    }
    for(int i = 1; i <= n; i++)
        b.push_back(d[i]);

    sort(b.begin(), b.end());
    b.erase(unique(b.begin(), b.end()), b.end());

    for(auto &it : b)
        id[it] = ++cnt;
    for(int i = 1; i <= n; i++)
        s[id[d[i]]]++;
    for(auto &it : b)
        for(auto &its : b)
            val[id[it]][id[its]] = (it ^ its) * (it | its) * (it & its);
    ll ans = 0;
    for(int i = 1; i <= cnt; i++)
        for(int j = i + 1; j <= cnt; j++)
            ans = (ans + ((val[i][j] * s[i] % mod) * s[j] % mod)) % mod;
    cout<<ans<<endl;
}
```

思维，模拟，很快得到思路，但样例2看了半天，最后队友读出要满足前缀和<span class="math inline">\(pre_n \geq 0\)</span>和<span class="math inline">\(a_1 \geq 0\)</span>，记维护前缀和的最大值<span class="math inline">\(\text{mav}\)</span>，对每个<span class="math inline">\(\text{resources} + a_i  &lt; 0\)</span>给它加上前缀和最大值<span class="math inline">\(\text{mav}\)</span>直到<span class="math inline">\(\text{resources} \geq 0\)</span>，我的第一发漏了考虑除0的情况，RE了一发



```cpp
typedef long long ll;

const int mod = 998244353;
const int N = 1e5 + 10;
int n;
ll a[N], s[N], mav[N];

void solve()
{
    cin>>n;
    for(int i = 1; i <= n; i++)
        cin>>a[i];
    for(int i = 1; i <= n; i++)
        s[i] = s[i - 1] + a[i];
    if(a[1] < 0 || s[n] < 0)
    {
        cout<<-1<<endl;
        return;
    }
    ll re = 0, ti = 0;
    for(int i = 1; i <= n; i++)
    {
        mav[i] = max(mav[i - 1], s[i]);
        if(re + s[i] >= 0)
            re += s[i], ti++;
        else
        {
            ll d = abs(re + s[i]);
            if(mav[i] == 0)
            {
                cout<<-1<<endl;
                return;
            }
            ll t = d / mav[i] + (d % mav[i] != 0);
            re += t * mav[i] + s[i], ti += t + 1;
        }
    }
    cout<<ti<<endl;
}
```

    int n;  cin>>n;

    {
        cin>>a[i];

    {
        op[i] = '|';
        if(a[i] == 1)
        {

            if(op[i - 1] == 'a')
                op[i + 1] = 'b';
            else
                op[i + 1] = 'a';
            i++;
        }
        else
        {
            for(int r = 1; r <= a[i] - 2; r++)
                op[i + r] = op[i - r];
            i = i + a[i] - 2;
        }
    }
    for(int i = 1; i <= 2 * n + 2; i++)
        if(op[i] == 'a' || op[i] == 'b')
            cout<<op[i];
    cout<<endl;
}

typedef long long ll;
const int mod = 998244353;

ll qmi(ll a, ll b)
{
    ll res = 1;
    while(b)
    {
        if(b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

void solve()
{
    ll n, m;   cin>>n>>m;
    for(ll x = 1; x <= m; x++)
    {
        ll p = qmi(m - x, n);
        ll q = qmi(m - 1, n);
        q = qmi(q, mod - 2);
        cout<<p * q % mod<<endl;
    }
}
