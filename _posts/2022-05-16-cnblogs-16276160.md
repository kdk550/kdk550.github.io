//  AC one more times

#include <iostream>
using namespace std;
int main()
{
    for(int v=1;v<=9;v++)
    {
        printf("%5c%5c %5c %5c %5c\n",'v','u','a','b','c');
        cout<<"~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~\n";
        for(int u=v+1;u<=10;u++)
        {
            int a=u*u-v*v,b=2*v*u,c=u*u+v*v;
            if(a*a+b*b==c*c)
                printf("%5d%5d( %5d,%5d,%5d)\n",v,u,a,b,c);
            printf("\n\n");
        }    
    }
    return 0;
}
